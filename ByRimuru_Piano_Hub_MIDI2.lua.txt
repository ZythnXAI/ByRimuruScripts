local function loadSettings()
    if isfile and readfile and isfile(SETTINGS_FILE) then
        local ok, content = pcall(function() return readfile(SETTINGS_FILE) end)
        if ok and content and content:find("talentlessSettings") then
            local fn = loadstring and loadstring(content)
            if fn then pcall(fn) end
        end
    end
end

local function applySettingsToConfig()
    local ts = _G.talentlessSettings
    if not ts then return end
    if ts.secondaryloader       ~= nil then Config.SecondaryLoader       = (ts.secondaryloader == true) end
    if ts.disablefeaturedsongs  ~= nil then Config.DisableFeaturedSongs  = (ts.disablefeaturedsongs == true) end
    if ts.alwaysshowmidispoofer ~= nil then Config.AlwaysShowMidiSpoofer = (ts.alwaysshowmidispoofer == true) end
    if ts.disableaccidents      ~= nil then Config.DisableAccidents      = (ts.disableaccidents == true) end
    if ts.errormargin           ~= nil then Config.ErrorMargin           = tonumber(ts.errormargin) or 0 end
    if ts.shorts                ~= nil then Config.ShortNotesEnabled     = (ts.shorts == true) end
    if ts.randomshift           ~= nil then Config.RandomShiftEnabled    = (ts.randomshift == true) end
end

local function saveSettings()
    if not writefile then return end
    _G.talentlessSettings.secondaryloader       = Config.SecondaryLoader
    _G.talentlessSettings.disablefeaturedsongs  = Config.DisableFeaturedSongs
    _G.talentlessSettings.alwaysshowmidispoofer = Config.AlwaysShowMidiSpoofer
    _G.talentlessSettings.disableaccidents      = Config.DisableAccidents
    _G.talentlessSettings.errormargin           = Config.ErrorMargin
    _G.talentlessSettings.shorts                = Config.ShortNotesEnabled
    _G.talentlessSettings.randomshift           = Config.RandomShiftEnabled
    local s = "_G.talentlessSettings = {\n"
    for key, val in pairs(_G.talentlessSettings) do
        if type(val) == "boolean" or type(val) == "number" then
            s = s .. "['" .. tostring(key) .. "'] = " .. tostring(val) .. ",\n"
        end
    end
    s = s .. "}"
    pcall(function() writefile(SETTINGS_FILE, s) end)
end

-- Escala cromática de 61 teclas + Mapeamento MIDI88 (notas MIDI 21 a 108 do loader_main.lua)
local PianoScale61 = "1!2@34$5%6^78*9(0qQwWeErtTyYuIoOpPasSdDfFgGhHjJkKlLzZxXcCvVbBnNm"
local CharToPitchIndex = {}
local midiKeyMappings  = {}
for idx = 1, #PianoScale61 do
    local ch = PianoScale61:sub(idx, idx)
    CharToPitchIndex[ch] = idx
    midiKeyMappings[ch]  = 35 + idx -- C2 (36) a C7 (96)
end
-- Extensão MIDI88 (Ctrl+tecla para oitavas extremas 21..35 e 97..108)
for idx = 1, 15 do
    local ch = PianoScale61:sub(idx, idx)
    midiKeyMappings["Ctrl+" .. ch] = 20 + idx
end

local function transposeChar(ch)
    local oct = Config.TransposeOctave or 0
    if oct == 0 or not ch then return ch end
    local idx = CharToPitchIndex[ch]
    if not idx then return ch end
    local newIdx = math.clamp(idx + (oct * 12), 1, #PianoScale61)
    return PianoScale61:sub(newIdx, newIdx)
end

-- Filtro de Mão Direita / Mão Esquerda para o Modo Dueto (testduetloader)
local function shouldPlayInDuetRole(ch)
    if State.duetRole == "Solo" or not ch then return true end
    local idx = CharToPitchIndex[ch]
    if not idx then return true end
    -- Ponto de divisão no Dó Central (C4 = índice 25, tecla 't')
    if State.duetRole == "Part1_High" then
        return idx >= 25
    elseif State.duetRole == "Part2_Low" then
        return idx < 25
    end
    return true
end

-- adjustVelocity (loader_main.lua): quando _G.midiSpoof está ativo, define currentVelocity
-- diretamente sem gastar eventos LeftAlt de teclado!
local function adjustVelocity(vel)
    if not State.isPlaying then return end
    vel = math.clamp(tonumber(vel) or 1.0, 0, 1)
    currentVelocity = vel
    if _G.midiSpoof or Config.MidiSpoofEnabled then
        _G.currentMidiVelocity = math.floor(vel * 127)
        return
    end
    local topress
    if vel < 0.27 then topress = "2"
    elseif vel >= 0.88 then topress = "c"
    else
        local index = math.clamp(math.floor((vel - 0.27) / 0.61 * (#velocityMap - 2)) + 2, 2, #velocityMap)
        topress = velocityMap:sub(index, index)
    end
    pcall(function()
        VIM:SendKeyEvent(true,  Enum.KeyCode.LeftAlt, false, game)
        VIM:SendKeyEvent(true,  KeyMap[topress], false, game)
        VIM:SendKeyEvent(false, KeyMap[topress], false, game)
        VIM:SendKeyEvent(false, Enum.KeyCode.LeftAlt, false, game)
    end)
end

-- ══════════════════════════════════════════════════════════════
--  IDENTIFICAÇÃO, FILTRAGEM E INTERPRETAÇÃO EXCLUSIVA MIDI 2
-- ══════════════════════════════════════════════════════════════
local function isMidi2LuaNative(seq)
    if not seq or type(seq) ~= "string" then return false end
    if seq:find("keypress%s*%(") or seq:find("pressnote%s*%(") then return true end
    if seq:find("Generated by MIDI2LUA") then return true end
    return false
end

local function extractMidi2LuaBPM(seq)
    if not seq then return nil end
    local b = seq:match("local%s+bpm%s*=%s*(%d+)") or seq:match("bpm%s*=%s*(%d+)")
    return b and tonumber(b) or nil
end

local function extractMidi2LuaFlags(seq)
    local flags = { shorts = true, velocity = false, midi88 = false }
    if not seq then return flags end
    if seq:find("Short Notes:%s*OFF") then flags.shorts = false
    elseif seq:find("Short Notes:%s*ON") then flags.shorts = true end
    if seq:find("Note Velocity:%s*ON") then flags.velocity = true end
    if seq:find("MIDI88:%s*ON") then flags.midi88 = true end
    return flags
end

local function analyzeSequence(seq)
    if not seq or type(seq) ~= "string" then
        return { dashes = 0, spaces = 0, notes = 0, len = 0, dashRatio = 0, spaceRatio = 0 }
    end
    local dashes, spaces, notes = 0, 0, 0
    local len = #seq
    for i = 1, len do
        local c = seq:sub(i, i)
        if c == "-" then dashes = dashes + 1
        elseif c == " " then spaces = spaces + 1
        elseif c ~= "[" and c ~= "]" then notes = notes + 1 end
    end
    return {
        dashes     = dashes,
        spaces     = spaces,
        notes      = notes,
        len        = len,
        dashRatio  = (len > 0) and (dashes / len) or 0,
        spaceRatio = (len > 0) and (spaces / len) or 0,
    }
end

local function isMidi2Song(song, stats)
    if not song or type(song.sequence) ~= "string" then return false end
    if song.custom == true then return true end
    if isMidi2LuaNative(song.sequence) then return true end
    if song.midi2lua == true then return true end

    local fmt = string.lower(tostring(song.format or song.source or ""))
    if fmt:find("midi2") or fmt:find("m2l") then return true end

    local cat = string.lower(tostring(song.category or ""))
    if cat:find("midi2") or cat:find("midi 2") or cat == "midi" then return true end

    local nm = string.lower(tostring(song.name or ""))
    if nm:find("midi2") or nm:find("midi 2") or nm:find("m2l") then return true end

    stats = stats or analyzeSequence(song.sequence)
    if stats.dashes >= 16 and stats.dashRatio >= 0.25 and stats.spaceRatio < 0.15 then
        return true
    end
    return false
end

local function isPlayableSeq(seq)
    if not seq or type(seq) ~= "string" then return false end
    if isMidi2LuaNative(seq) then return true end
    local n = 0
    for i = 1, #seq do
        local c = seq:sub(i, i)
        if KeyMap[c] then
            n = n + 1
            if n >= 3 then return true end
        end
    end
    return false
end

local function cleanSongName(rawName)
    local s = tostring(rawName or "Sem Título")
    s = s:gsub("%.[mM][iI][dD][iI]?$", "")
    s = s:gsub("%.[lL][uU][aA]$", "")
    s = s:gsub("%.[tT][xX][tT]$", "")
    s = s:gsub("%b[]", function(m)
        local low = m:lower()
        if low:find("midi") or low:find("m2l") then return "" end
        return m
    end)
    s = s:gsub("%b()", function(m)
        local low = m:lower()
        if low:find("midi") or low:find("m2l") then return "" end
        return m
    end)
    s = s:gsub("[%-–—]%s*[mM][iI][dD][iI]%s*2?[lL]?[uU]?[aA]?$", "")
    s = s:gsub("_+", " ")
    s = s:gsub("%(%s*%)", "")
    s = s:gsub("%[%s*%]", "")
    s = s:gsub("%s+", " ")
    s = s:match("^%s*(.-)%s*$") or s
    if s == "" then s = "Track MIDI 2" end
    return s
end

local function normalizeSongKey(cleanName)
    return string.lower(cleanName):gsub("[%s%p]+", "")
end

local function prepareSong(song)
    if not song then return song end
    local stats = analyzeSequence(song.sequence)
    local native = isMidi2LuaNative(song.sequence)
    local bpm = tonumber(song.bpm) or 0

    song.name             = cleanSongName(song.name)
    song._stats           = stats
    song._midi2lua        = true
    song._midi2lua_native = native

    if native then
        local scriptBPM = extractMidi2LuaBPM(song.sequence)
        if bpm <= 0 then
            bpm = (scriptBPM and scriptBPM > 0) and scriptBPM or Config.DefaultBPM
        end
        song._m2l_flags = extractMidi2LuaFlags(song.sequence)
    else
        if bpm <= 0 then
            bpm = Config.DefaultBPM
        end
    end

    song._play_bpm   = bpm
    song._speed_mult = tonumber(song.speed_mult) or 1.0
    song._prepared   = true
    return song
end

local PlayableIdx    = {}
local PlayableLookup = {}
local CategoryCount  = {}

local function processAndFilterSongs()
    Songs          = {}
    PlayableIdx    = {}
    PlayableLookup = {}
    CategoryCount  = {}

    local seenKeys      = {}
    local keptMidi2     = 0
    local removedLegacy = 0
    local removedDupes  = 0

    local function considerSong(raw)
        if not raw or not isPlayableSeq(raw.sequence) then
            removedLegacy = removedLegacy + 1
            return
        end
        if Config.DisableFeaturedSongs and string.lower(tostring(raw.category or "")) == "featured" then
            return
        end
        local stats = analyzeSequence(raw.sequence)
        if Config.Midi2Exclusive and not isMidi2Song(raw, stats) then
            removedLegacy = removedLegacy + 1
            return
        end

        local cleanedName = cleanSongName(raw.name)
        local normKey     = normalizeSongKey(cleanedName)

        if Config.DeduplicateSongs and normKey ~= "" and seenKeys[normKey] then
            local existingIdx = seenKeys[normKey]
            local existing = Songs[existingIdx]
            if existing and (#tostring(raw.sequence) > #tostring(existing.sequence)) then
                raw.name = cleanedName
                prepareSong(raw)
                Songs[existingIdx] = raw
            end
            removedDupes = removedDupes + 1
            return
        end

        local entry = {
            name       = cleanedName,
            bpm        = tonumber(raw.bpm) or 0,
            emoji      = "♪",
            category   = (raw.category and raw.category ~= "") and tostring(raw.category) or "MIDI 2",
            sequence   = raw.sequence,
            custom     = raw.custom == true,
            midi2lua   = true,
            speed_mult = raw.speed_mult,
        }
        prepareSong(entry)

        table.insert(Songs, entry)
        local newIdx = #Songs
        if normKey ~= "" then
            seenKeys[normKey] = newIdx
        end
        keptMidi2 = keptMidi2 + 1
    end

    State.stats.totalRaw = #RawSongs
    for i = 1, #RawSongs do
        considerSong(RawSongs[i])
    end

    for _, c in ipairs(State.customSongs) do
        considerSong({
            name     = c.name,
            bpm      = c.bpm or 120,
            category = "Custom",
            sequence = c.sequence,
            custom   = true,
            midi2lua = true,
        })
    end

    for i = 1, #Songs do
        local s = Songs[i]
        if s then
            table.insert(PlayableIdx, i)
            PlayableLookup[i] = #PlayableIdx
            local cat = s.category or "MIDI 2"
            CategoryCount[cat] = (CategoryCount[cat] or 0) + 1
        end
    end

    State.stats.keptMidi2     = keptMidi2
    State.stats.removedLegacy = removedLegacy
    State.stats.removedDupes  = removedDupes
end

-- ══════════════════════════════════════════════════════════════
--  PARSERS MIDI 2 (INCLUINDO MERGE DE ACORDES E MIDI88 CTRL+)
-- ══════════════════════════════════════════════════════════════
local function parseMidi2GridSequence(seq)
    local cleaned = (seq or ""):gsub("^[-%s]+", ""):gsub("[-%s]+$", "")
    local tokens, i, len = {}, 1, #cleaned
    while i <= len do
        local ch = cleaned:sub(i, i)
        if ch == "[" then
            local j = cleaned:find("]", i)
            if j then
                local keys = {}
                for k = i + 1, j - 1 do
                    local c = cleaned:sub(k, k)
                    if KeyMap[c] then
                        keys[#keys + 1] = { key = KeyMap[c], shift = ShiftKeys[c] or false, char = c }
                    end
                end
                if #keys > 0 then
                    tokens[#tokens + 1] = { type = "chord", keys = keys }
                end
                i = j + 1
            else
                i = i + 1
            end
        elseif ch == "-" then
            tokens[#tokens + 1] = { type = "dash" }
            i = i + 1
        elseif KeyMap[ch] then
            tokens[#tokens + 1] = { type = "note", key = KeyMap[ch], shift = ShiftKeys[ch] or false, char = ch }
            i = i + 1
        else
            i = i + 1
        end
    end
    return tokens
end

-- Parser MIDI2LUA Nativo (loader_main.lua):
-- Suporta mergeCount (funde keypress consecutivos de mesma duração em acordes limpos)
-- e suporta modificadores "Ctrl+" do modo MIDI88!
local function parseMidi2LuaScript(seq)
    local actions = {}
    if not seq then return actions end
    local cleaned = seq:gsub("%-%-[^\r\n]*", " ")

    for fnName, argsStr in cleaned:gmatch("([a-zA-Z_][a-zA-Z0-9_]*)%s*(%b())") do
        local inner = argsStr:sub(2, -2)
        if fnName == "keypress" then
            local keys, beats = inner:match('^%s*["\']([^"\']+)["\']%s*,%s*([^,%s)]+)')
            if keys then
                local b = tonumber(beats)
                local hasCtrl = keys:find("Ctrl%+") ~= nil
                local prev = actions[#actions]
                -- Otimização oficial do loader_main.lua: funde keypress consecutivos sem rest entre eles
                if prev and prev.type == "keypress" and prev.beats == b and not prev.hasCtrl and not hasCtrl then
                    prev.keys = prev.keys .. keys
                    prev.mergeCount = (prev.mergeCount or 1) + 1
                else
                    actions[#actions + 1] = {
                        type       = "keypress",
                        keys       = keys,
                        beats      = b,
                        isShort    = (b == nil),
                        hasCtrl    = hasCtrl,
                        mergeCount = 1,
                    }
                end
            end
        elseif fnName == "pressnote" then
            local note, octave, pbeats = inner:match('^%s*["\']([^"\']+)["\']%s*,%s*(%d+)%s*,%s*([^,%s)]+)')
            if note then
                local b = tonumber(pbeats)
                actions[#actions + 1] = {
                    type    = "pressnote",
                    note    = note,
                    octave  = tonumber(octave),
                    beats   = b,
                    isShort = (b == nil),
                }
            end
        elseif fnName == "rest" then
            local rbeats = inner:match("^%s*([%d%.]+)")
            if rbeats then
                actions[#actions + 1] = {
                    type  = "rest",
                    beats = tonumber(rbeats) or 0.25,
                }
            end
        elseif fnName == "adjustVelocity" then
            local vel = inner:match("^%s*([%d%.]+)")
            if vel then
                actions[#actions + 1] = {
                    type = "adjustVelocity",
                    vel  = tonumber(vel) or 1.0,
                }
            end
        elseif fnName == "pedalDown" then
            actions[#actions + 1] = { type = "pedalDown" }
        elseif fnName == "pedalUp" then
            actions[#actions + 1] = { type = "pedalUp" }
        end
    end
    return actions
end

local MouseMap = nil

local function isHubObj(o)
    local cur = o
    while cur and cur ~= game do
        local n = tostring(cur.Name):lower()
        if n:find("rimuru") or n:find("talentless") then return true end
        cur = cur.Parent
    end
    return false
end

local function buildMouseMap()
    MouseMap = {}
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then return end
    for _, o in ipairs(pg:GetDescendants()) do
        if (o:IsA("TextButton") or o:IsA("ImageButton")) and not isHubObj(o) then
            local id
            local nm = tostring(o.Name)
            if #nm == 1 then id = nm
            else
                local t = ""
                pcall(function() t = tostring(o.Text or "") end)
                if #t == 1 then id = t end
            end
            if id and not MouseMap[id] then MouseMap[id] = o end
        end
    end
end

local function clickKeyChar(char, holdTime)
    if not char then return false end
    if not MouseMap then buildMouseMap() end
    local btn = MouseMap and (MouseMap[char] or MouseMap[char:lower()] or MouseMap[char:upper()])
    if not btn or not btn.Parent then return false end
    local p, s = btn.AbsolutePosition, btn.AbsoluteSize
    local x, y = math.floor(p.X + s.X / 2), math.floor(p.Y + s.Y / 2)
    pcall(function() VIM:SendMouseButtonEvent(x, y, 0, true, game, 0) end)
    task.delay(holdTime or Config.KeyHold, function()
        pcall(function() VIM:SendMouseButtonEvent(x, y, 0, false, game, 0) end)
    end)
    return true
end

-- ══════════════════════════════════════════════════════════════
--  ENVIO DE TECLAS — ASSÍNCRONO + SUPORTE A MIDI88 & DUETO
-- ══════════════════════════════════════════════════════════════
local function preciseWait(seconds)
    if seconds <= 0 then return end
    local start = os.clock()
    while os.clock() - start < seconds do
        if not State.isPlaying then return end
        if seconds - (os.clock() - start) > 0.014 then
            RunService.Heartbeat:Wait()
        else
            break
        end
    end
    while os.clock() - start < seconds do
        if not State.isPlaying then return end
    end
end

local function calcHoldTime(shift, isShort, explicitHoldSec)
    if explicitHoldSec and explicitHoldSec > 0 then
        return math.clamp(explicitHoldSec, 0.02, 2.50)
    end
    if isShort and Config.ShortNotesEnabled then
        if shift then
            return Config.ShortMinHoldShift + math.random() * (Config.ShortMaxHoldShift - Config.ShortMinHoldShift)
        else
            return Config.ShortMinHold + math.random() * (Config.ShortMaxHold - Config.ShortMinHold)
        end
    end
    return Config.KeyHold
end

local function scheduleKeyRelease(keyCode, shift, ctrl, holdTime)
    local token = os.clock() + math.random()
    ActiveKeyTokens[keyCode] = token
    task.spawn(function()
        if Config.SecondaryLoader then
            preciseWait(holdTime)
        else
            task.wait(holdTime)
        end
        if ActiveKeyTokens[keyCode] == token then
            ActiveKeyTokens[keyCode] = nil
            pcall(function()
                VIM:SendKeyEvent(false, keyCode, false, game)
                if shift then VIM:SendKeyEvent(false, Enum.KeyCode.LeftShift, false, game) end
                if ctrl  then VIM:SendKeyEvent(false, Enum.KeyCode.LeftControl, false, game) end
            end)
        end
    end)
end

local function pressKey(keyCode, shift, char, isShort, explicitHoldSec, ctrl)
    if not State.isPlaying then return end
    local actualChar  = transposeChar(char) or char
    if not shouldPlayInDuetRole(actualChar) then return end

    local actualKey   = KeyMap[actualChar] or keyCode
    local actualShift = (ShiftKeys[actualChar] ~= nil) and ShiftKeys[actualChar] or shift
    local holdTime    = calcHoldTime(actualShift, isShort, explicitHoldSec)

    if Config.MouseClickEnabled and clickKeyChar(actualChar, holdTime) then return end

    if not Config.DisableAccidents and not actualShift and Config.RandomShiftEnabled and Config.ErrorMargin > 0 then
        if math.random(1, 500) <= math.floor(Config.ErrorMargin * 100) then
            actualShift = true
        end
    end

    pcall(function()
        if ctrl        then VIM:SendKeyEvent(true, Enum.KeyCode.LeftControl, false, game) end
        if actualShift then VIM:SendKeyEvent(true, Enum.KeyCode.LeftShift, false, game) end
        VIM:SendKeyEvent(true, actualKey, false, game)
    end)

    scheduleKeyRelease(actualKey, actualShift, ctrl, holdTime)
end

local function playChord(keys, isShort, explicitHoldSec)
    if not State.isPlaying or not keys or #keys == 0 then return end
    local holdTime = calcHoldTime(true, isShort, explicitHoldSec)

    local kb = {}
    for _, k in ipairs(keys) do
        local tc = transposeChar(k.char) or k.char
        if shouldPlayInDuetRole(tc) then
            local tk = KeyMap[tc] or k.key
            local ts = (ShiftKeys[tc] ~= nil) and ShiftKeys[tc] or k.shift
            if not (Config.MouseClickEnabled and clickKeyChar(tc, holdTime)) then
                kb[#kb + 1] = { key = tk, shift = ts, char = tc }
            end
        end
    end
    if #kb == 0 then return end

    local nonShift, withShift = {}, {}
    for _, k in ipairs(kb) do
        if k.shift then withShift[#withShift + 1] = k
        else nonShift[#nonShift + 1] = k end
    end

    local token = os.clock() + math.random()
    pcall(function()
        for _, k in ipairs(nonShift) do
            ActiveKeyTokens[k.key] = token
            VIM:SendKeyEvent(true, k.key, false, game)
        end
        if #withShift > 0 then
            VIM:SendKeyEvent(true, Enum.KeyCode.LeftShift, false, game)
            for _, k in ipairs(withShift) do
                ActiveKeyTokens[k.key] = token
                VIM:SendKeyEvent(true, k.key, false, game)
            end
        end
    end)

    task.spawn(function()
        if Config.SecondaryLoader then
            preciseWait(holdTime)
        else
            task.wait(holdTime)
        end
        pcall(function()
            if #withShift > 0 then
                for _, k in ipairs(withShift) do
                    if ActiveKeyTokens[k.key] == token then
                        ActiveKeyTokens[k.key] = nil
                        VIM:SendKeyEvent(false, k.key, false, game)
                    end
                end
                VIM:SendKeyEvent(false, Enum.KeyCode.LeftShift, false, game)
            end
            for _, k in ipairs(nonShift) do
                if ActiveKeyTokens[k.key] == token then
                    ActiveKeyTokens[k.key] = nil
                    VIM:SendKeyEvent(false, k.key, false, game)
                end
            end
        end)
    end)
end

local function keypressStr(keys, isShort, explicitHoldSec)
    if not keys or keys == "" then return end
    -- Suporte ao prefixo "Ctrl+" do modo MIDI88 (loader_main.lua)
    if keys:sub(1, 5) == "Ctrl+" and Config.Midi88Enabled then
        local ch = keys:sub(6, 6)
        if KeyMap[ch] then
            pressKey(KeyMap[ch], ShiftKeys[ch] or false, ch, isShort, explicitHoldSec, true)
        end
        return
    end
    local list = {}
    for i = 1, #keys do
        local c = keys:sub(i, i)
        if KeyMap[c] then
            list[#list + 1] = { key = KeyMap[c], shift = ShiftKeys[c] or false, char = c }
        end
    end
    if #list == 1 then
        pressKey(list[1].key, list[1].shift, list[1].char, isShort, explicitHoldSec, false)
    elseif #list > 1 then
        playChord(list, isShort, explicitHoldSec)
    end
end

-- ══════════════════════════════════════════════════════════════
--  MOTOR DE RITMO — AGENDAMENTO ABSOLUTO + CONTROLES LILGUI
--  Inclui upbpm/downbpm (±10 BPM em tempo real) e scrubbing (Seek)
-- ══════════════════════════════════════════════════════════════
local function getEffectiveSpeed()
    local sm = State.speed
    if State.mode == "Nightcore" then sm = sm * Config.NightcoreSpeed
    elseif State.mode == "Turbo" then sm = sm * Config.TurboSpeed
    elseif State.mode == "Slowed" then sm = sm * Config.SlowedSpeed end
    return math.max(0.1, sm)
end

local function getActiveSongBPM(song)
    local base = (song and (song._play_bpm or song.bpm)) or Config.DefaultBPM
    return math.clamp(base + (State.bpmOffset or 0), 20, 480)
end

local function baseStep(song)
    if song and not song._prepared then prepareSong(song) end
    local bpm  = getActiveSongBPM(song)
    local mult = (song and (song._speed_mult or song.speed_mult)) or 1.0
    local step = (60 / bpm) * (1 / Config.SubdivPerBeat) / mult
    local d    = step / getEffectiveSpeed()
    if Config.AntiLagEnabled and d < Config.AntiLagThrottle then d = Config.AntiLagThrottle end
    return d
end

local function playSong(idx)
    local song = Songs[idx]
    if not song then return end
    if not song._prepared then prepareSong(song) end
    releaseAllKeys()

    -- ═══ 1. EXECUÇÃO MIDI 2 NATIVA (keypress / pressnote / rest) ═══
    if song._midi2lua_native then
        local actions = parseMidi2LuaScript(song.sequence)
        local flags   = song._m2l_flags or { shorts = true, velocity = false }
        State.totalNotes  = #actions
        State.currentNote = 0
        State.progress    = 0
        State.seekTarget  = nil
        if #actions == 0 then return end

        local startClock  = os.clock()
        local virtualTime = 0
        local ai          = 1

        while ai <= #actions and State.isPlaying do
            while State.isPaused do
                releaseAllKeys()
                task.wait(0.03)
                if not State.isPlaying then return end
                startClock = os.clock() - virtualTime
            end

            -- Playhead Scrubbing (Seek da barra de progresso do lilgui)
            if State.seekTarget then
                local targetIdx = math.clamp(math.floor(State.seekTarget * #actions), 1, #actions)
                State.seekTarget = nil
                releaseAllKeys()
                ai = targetIdx
                virtualTime = 0
                startClock = os.clock()
            end

            local action       = actions[ai]
            local bpm          = getActiveSongBPM(song)
            local effSpeed     = getEffectiveSpeed()
            local isShort      = flags.shorts and (action.isShort ~= false)
            local explicitHold = nil

            if action.beats and type(action.beats) == "number" then
                isShort = false
                explicitHold = (action.beats / bpm) * 60 / effSpeed
            end

            if action.type == "keypress" then
                keypressStr(action.keys, isShort, explicitHold)
            elseif action.type == "pressnote" then
                local map = noteMappings[action.note]
                if map and map[action.octave] then
                    local ch = map[action.octave]
                    local kc = KeyMap[ch]
                    if kc then
                        pressKey(kc, ShiftKeys[ch] or false, ch, isShort, explicitHold, false)
                    end
                end
            elseif action.type == "rest" then
                local waitTime = ((action.beats or 0.25) / bpm) * 60 / effSpeed
                virtualTime = virtualTime + waitTime
                local jit = 0
                if Config.ErrorMargin > 0 or Config.HumanizeEnabled then
                    local m = math.max(Config.ErrorMargin, Config.HumanizeEnabled and Config.HumanizeJitter or 0)
                    jit = (math.random() * 2 - 1) * (m * 0.5)
                end
                local target = startClock + virtualTime + jit
                local rem = target - os.clock()
                if rem > 0 then
                    preciseWait(rem)
                end
            elseif action.type == "adjustVelocity" then
                adjustVelocity(action.vel)
            elseif action.type == "pedalDown" then
                pedalDown()
            elseif action.type == "pedalUp" then
                pedalUp()
            end

            State.currentNote = ai
            State.progress    = ai / #actions
            ai = ai + 1

            if ai % 32 == 0 then
                RunService.Heartbeat:Wait()
            end
        end
        releaseAllKeys()
        return
    end

    -- ═══ 2. EXECUÇÃO MIDI 2 EM GRADE (Grid) ═══
    local tokens = parseMidi2GridSequence(song.sequence)
    State.totalNotes  = #tokens
    State.currentNote = 0
    State.progress    = 0
    State.seekTarget  = nil
    if #tokens == 0 then return end

    local startClock  = os.clock()
    local virtualTime = 0
    local catchUp     = 0
    local ti          = 1

    while ti <= #tokens and State.isPlaying do
        while State.isPaused do
            releaseAllKeys()
            task.wait(0.03)
            if not State.isPlaying then return end
            startClock = os.clock() - virtualTime
        end

        if State.seekTarget then
            local targetIdx = math.clamp(math.floor(State.seekTarget * #tokens), 1, #tokens)
            State.seekTarget = nil
            releaseAllKeys()
            ti = targetIdx
            virtualTime = 0
            startClock = os.clock()
        end

        local token        = tokens[ti]
        local step         = baseStep(song)
        local nextTok      = tokens[ti + 1]
        local hasDashAfter = (nextTok and nextTok.type == "dash")
        local isShort      = Config.ShortNotesEnabled

        local fired = false
        if token.type == "note" then
            pressKey(token.key, token.shift, token.char, isShort, nil, false)
            if not hasDashAfter then
                virtualTime = virtualTime + step
            end
            fired = true
        elseif token.type == "chord" then
            playChord(token.keys, isShort, nil)
            if not hasDashAfter then
                virtualTime = virtualTime + step
            end
            fired = true
        elseif token.type == "dash" then
            virtualTime = virtualTime + step
        end

        local jit = 0
        if fired and (Config.HumanizeEnabled or Config.ErrorMargin > 0) then
            local m = math.max(Config.ErrorMargin, Config.HumanizeEnabled and Config.HumanizeJitter or 0)
            jit = (math.random() * 2 - 1) * m
        end

        State.currentNote = ti
        State.progress    = ti / #tokens

        local target = startClock + virtualTime + jit
        local now    = os.clock()
        if now < target then
            catchUp = 0
            local remaining = target - now
            if Config.SecondaryLoader then
                preciseWait(remaining)
            else
                if remaining > 0.03 then task.wait(remaining - 0.012) end
                while os.clock() < target do
                    if not State.isPlaying then break end
                    RunService.Heartbeat:Wait()
                end
            end
        else
            catchUp = catchUp + 1
            if catchUp >= 24 then
                catchUp = 0
                RunService.Heartbeat:Wait()
            end
        end

        ti = ti + 1
    end
    releaseAllKeys()
end

local refreshNowPlaying, refreshList

local function getRandomPlayable()
    if #PlayableIdx == 0 then return 1 end
    return PlayableIdx[math.random(1, #PlayableIdx)]
end

local function stopPlayback()
    State.isPlaying       = false
    State.isPaused        = false
    State.duetSyncWaiting = false
    State.progress        = 0
    State.currentNote     = 0
    State.seekTarget      = nil
    releaseAllKeys()
    if State.playThread then
        pcall(function() task.cancel(State.playThread) end)
        State.playThread = nil
    end
end

local function startPlayback()
    if State.isPlaying or #Songs == 0 then return end
    State.isPlaying = true
    State.isPaused  = false
    State.playThread = task.spawn(function()
        repeat
            playSong(State.currentSong)
            if State.isPlaying then
                if not State.autoNext and not State.loopEnabled then
                    State.isPlaying = false
                elseif State.loopEnabled then
                    -- repete a mesma música
                elseif #State.queue > 0 then
                    State.queuePos = (State.queuePos % #State.queue) + 1
                    State.currentSong = State.queue[State.queuePos]
                elseif State.shuffleEnabled then
                    State.currentSong = getRandomPlayable()
                else
                    State.currentSong = (State.currentSong % #Songs) + 1
                end
                if State.isPlaying then
                    if refreshNowPlaying then pcall(refreshNowPlaying) end
                    if refreshList then pcall(refreshList) end
                end
            end
        until not State.isPlaying
        State.isPlaying = false
        releaseAllKeys()
        if refreshNowPlaying then pcall(refreshNowPlaying) end
    end)
end

local function pauseResume()
    if not State.isPlaying then
        startPlayback()
    else
        State.isPaused = not State.isPaused
        if State.isPaused then releaseAllKeys() end
    end
end

local function nextSong()
    if #Songs == 0 then return end
    stopPlayback()
    State.bpmOffset = 0
    if #State.queue > 0 then
        State.queuePos = (State.queuePos % #State.queue) + 1
        State.currentSong = State.queue[State.queuePos]
    elseif State.shuffleEnabled then
        State.currentSong = getRandomPlayable()
    else
        State.currentSong = (State.currentSong % #Songs) + 1
    end
end

local function prevSong()
    if #Songs == 0 then return end
    stopPlayback()
    State.bpmOffset = 0
    State.currentSong = ((State.currentSong - 2) % #Songs) + 1
end

local function playIndex(idx, autoNext)
    if not Songs[idx] then return end
    stopPlayback()
    State.bpmOffset   = 0
    State.currentSong = idx
    if autoNext ~= nil then State.autoNext = autoNext and true or false end
    startPlayback()
end

loadData()
processAndFilterSongs()
if not Songs[State.currentSong] then
    State.currentSong = PlayableIdx[1] or 1
end

local function hoverify(btn, normalColor, hoverColor)
    local activeTween
    local function tweenTo(c)
        if activeTween then pcall(function() activeTween:Cancel() end) end
        activeTween = TweenService:Create(btn, TweenInfo.new(0.14, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            BackgroundColor3 = c
        })
        activeTween:Play()
    end
    btn.MouseEnter:Connect(function() tweenTo(hoverColor) end)
    btn.MouseLeave:Connect(function() tweenTo(normalColor) end)
end

local MAX_LIST = 75

local function buildFilteredIndices()
    local out = {}
    local q = string.lower(SearchBox.Text or "")
    local function passSearch(s)
        if q == "" then return true end
        return (string.find(string.lower(s.name or ""), q, 1, true) ~= nil)
            or (string.find(string.lower(s.category or ""), q, 1, true) ~= nil)
    end

    if activeCategory == "★ Favoritos" then
        for i = 1, #Songs do
            local s = Songs[i]
            if s and isFav(i) and passSearch(s) then out[#out + 1] = i end
        end
    elseif activeCategory == "Custom" then
        for i = 1, #Songs do
            local s = Songs[i]
            if s and s.custom and passSearch(s) then out[#out + 1] = i end
        end
    else
        for _, i in ipairs(PlayableIdx) do
            local s = Songs[i]
            if s and (activeCategory == nil or s.category == activeCategory) and passSearch(s) then
                out[#out + 1] = i
            end
        end
    end
    return out
end

local function buildCategories()
    for _, c in ipairs(CatScroll:GetChildren()) do
        if c:IsA("TextButton") then c:Destroy() end
    end
    local order = 1
    addCatBtn(tr("all"), nil, #PlayableIdx, order); order = order + 1
    addCatBtn(tr("favs"), "★ Favoritos", nil, order); order = order + 1

    local cc = 0
    for i = 1, #Songs do
        if Songs[i] and Songs[i].custom then cc = cc + 1 end
    end
    addCatBtn(tr("custom"), "Custom", cc, order); order = order + 1

    local sorted = {}
    for cat, n in pairs(CategoryCount) do
        sorted[#sorted + 1] = { cat = cat, n = n }
    end
    table.sort(sorted, function(a, b) return a.n > b.n end)
    for _, e in ipairs(sorted) do
        if e.cat ~= "Custom" then
            addCatBtn(e.cat, e.cat, e.n, order)
            order = order + 1
        end
    end
end

local function walkTranslate()
    local map = trMap()
    local i = langIdx()
    for _, o in ipairs(ScreenGui:GetDescendants()) do
        if o:IsA("TextButton") or o:IsA("TextLabel") then
            local base = o:GetAttribute("trbase")
            if not base and map[o.Text] then
                base = o.Text
                o:SetAttribute("trbase", base)
            end
            if base and map[base] then o.Text = map[base][i] end
        elseif o:IsA("TextBox") then
            local base = o:GetAttribute("trbasePH")
            if not base and map[o.PlaceholderText] then
                base = o.PlaceholderText
                o:SetAttribute("trbasePH", base)
            end
            if base and map[base] then o.PlaceholderText = map[base][i] end
        end
    end
end

local function applyLang()
    walkTranslate()
    buildCategories()
    setCategory(activeCategory)
    refreshList()
    refreshNowPlaying()
end

local function fullCleanup()
    stopPlayback()
    releaseAllKeys()
    for _, c in ipairs(Connections) do
        pcall(function() c:Disconnect() end)
    end
    Connections = {}
    pcall(function() ScreenGui:Destroy() end)
end

_G.ByRimuruCleanup = fullCleanup

CloseBtn.MouseButton1Click:Connect(fullCleanup)
ScreenGui.AncestryChanged:Connect(function(_, parent)
    if not parent then fullCleanup() end
end)

-- ══════════════════════════════════════════════════════════════
--  ARRASTAR JANELA E SCRUBBING DA BARRA DE PROGRESSO
-- ══════════════════════════════════════════════════════════════
do
    local dragging, dragStart, startPos = false, nil, nil
    TopBar.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging  = true
            dragStart = i.Position
            startPos  = Root.Position
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    trackConn(UserInputService.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            local d = i.Position - dragStart
            Root.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end))
end

trackConn(UserInputService.InputChanged:Connect(function(i)
    if seekingBar and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
        updateSeekFromInput(i.Position.X)
    end
end))
trackConn(UserInputService.InputEnded:Connect(function(i)
    if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
        seekingBar = false
    end
end))

-- ══════════════════════════════════════════════════════════════
--  ATUALIZAÇÃO DA BARRA DE PROGRESSO E ATALHO RIGHTALT
-- ══════════════════════════════════════════════════════════════
trackConn(RunService.RenderStepped:Connect(function()
    local target = State.isPlaying and State.progress or 0
    if not seekingBar then
        local cur = UI.progFill.Size.X.Scale
        UI.progFill.Size = UDim2.new(cur + (target - cur) * 0.25, 0, 1, 0)
    end
    if State.isPlaying then
        UI.playPauseBtn.Text = State.isPaused and "▶" or "⏸"
        local s = curSong()
        UI.nowText.Text = (tr("note")):format(
            s and s.name or "", math.floor(State.progress * 100), State.currentNote, State.totalNotes
        )
    else
        UI.playPauseBtn.Text = "⏯"
        if UI.progFill.Size.X.Scale < 0.01 then
            UI.nowText.Text = tr("ready")
        end
    end
end))

trackConn(UserInputService.InputBegan:Connect(function(i, gpe)
    if gpe then return end
    if i.KeyCode == Enum.KeyCode.RightAlt then
        if State.isMinimized or not Root.Visible then
            restoreWindow()
        else
            minimizeWindow()
        end
    end
end))

    end)
end
